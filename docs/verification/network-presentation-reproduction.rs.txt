    #[test]
    fn investigate_network_presentation_clock_and_control_publication() {
        let mut source = crate::demo::create_demo_world(1, None);
        let builder = source.simulation.builder_for_player(PlayerId(0)).unwrap();
        let command = PlayerCommand::MoveBuilder {
            builder: builder.id,
            destination: SimPoint::new(
                builder.position.x + 20 * castle_fight_sim::SUBUNITS_PER_WORLD_UNIT,
                builder.position.y,
            ),
        };
        let mut driver = MatchDriver::new(&source.simulation, source.content);
        driver.submit_local_command(&source.simulation, PlayerId(0), command);
        let result = driver.advance_local_tick(&mut source.simulation).unwrap();
        let demo = crate::demo::create_demo_world(4, None);
        let client = NetworkClient::connected_test_fixture(compatibility_identity_for_demo(&demo));
        let mut samples = PresentationSamples::new(PresentationSnapshot::capture(&demo.simulation));
        let mut selected = SelectedMatch {
            content: demo.content,
            direct_buildings: demo.direct_buildings,
            local_player: PlayerId(0),
        };
        let mut authoritative = AuthoritativeSimulation::new_networked(
            demo.simulation,
            demo.content,
            demo.match_config,
            client,
            PlayerId(0),
            0,
        );
        let AuthorityMode::Network { client, .. } = &authoritative.authority else {
            unreachable!()
        };
        client.inject_server_message_for_test(ServerMessage::StreamRecord {
            record: (&CanonicalStreamRecord::Tick(result.finalized.clone())).into(),
        });
        client.inject_server_message_for_test(ServerMessage::TickExecutions {
            batch: castle_fight_protocol::WireExecutionBatch {
                tick: result.finalized.tick,
                executions: result.executions.iter().copied().map(Into::into).collect(),
            },
        });
        process_network_events(&mut authoritative, &mut samples, &mut selected, None).unwrap();
        let previous = samples.previous.builders[&builder.id].position;
        let current = samples.current.builders[&builder.id].position;
        assert_ne!(previous, current);
        let mut fixed = Time::<Fixed>::from_hz(f64::from(CASTLE_FIGHT_SIMULATION_HZ));
        fixed.accumulate_overstep(Duration::from_millis(30));
        let before_alpha = SimulationPlayback::default().interpolation_alpha(&fixed);
        let before = presentation::sim_point_to_world_lerp(previous, current, before_alpha);
        fixed.accumulate_overstep(Duration::from_millis(6));
        fixed.discard_overstep(fixed.timestep());
        process_network_events(&mut authoritative, &mut samples, &mut selected, None).unwrap();
        let after_alpha = SimulationPlayback::default().interpolation_alpha(&fixed);
        let after = presentation::sim_point_to_world_lerp(
            samples.previous.builders[&builder.id].position,
            samples.current.builders[&builder.id].position,
            after_alpha,
        );
        println!(
            "empty network poll: samples={}->{} alpha={before_alpha:.3}->{after_alpha:.3} position.x={:.3}->{:.3}",
            samples.previous.tick, samples.current.tick, before.x, after.x
        );
        assert!(
            after.x < before.x,
            "reproduces a backwards render step during constant authoritative forward motion"
        );
        let control = driver
            .emit_local_control(
                &mut source.simulation,
                castle_fight_sim::MatchControlEvent::SetPlayerConnection {
                    player: PlayerId(0),
                    connection: castle_fight_sim::PlayerConnectionStatus::Connected,
                },
            )
            .unwrap();
        let AuthorityMode::Network { client, .. } = &authoritative.authority else {
            unreachable!()
        };
        client.inject_server_message_for_test(ServerMessage::StreamRecord {
            record: (&CanonicalStreamRecord::Control(control)).into(),
        });
        process_network_events(&mut authoritative, &mut samples, &mut selected, None).unwrap();
        println!(
            "non-tick control: samples={}->{} authoritative tick={}",
            samples.previous.tick,
            samples.current.tick,
            authoritative.simulation.tick()
        );
        assert_eq!(
            samples.previous.tick, samples.current.tick,
            "reproduces interpolation-history collapse without advancing the simulation"
        );
    }
